cmake_minimum_required(VERSION 3.24)
project(azrael_virtual_input LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This prototype requires Windows MSVC x64")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
add_library(minhook STATIC vendor/minhook/src/buffer.c vendor/minhook/src/hook.c
  vendor/minhook/src/trampoline.c vendor/minhook/src/hde/hde64.c)
target_include_directories(minhook PUBLIC vendor/minhook/include)
add_library(azrael-virtual-input-hook SHARED hook.cpp)
target_link_libraries(azrael-virtual-input-hook PRIVATE minhook user32 gdi32)
add_library(virtual-input-core STATIC broker.cpp)
target_include_directories(virtual-input-core PUBLIC vendor/json/single_include)
target_link_libraries(virtual-input-core PUBLIC user32 advapi32)
add_executable(azrael-virtual-input main.cpp)
target_link_libraries(azrael-virtual-input PRIVATE virtual-input-core)
add_executable(azrael-virtual-input-fixture WIN32 fixture.cpp)
target_link_libraries(azrael-virtual-input-fixture PRIVATE user32 comctl32)
add_executable(azrael-virtual-input-selftest selftest.cpp)
target_link_libraries(azrael-virtual-input-selftest PRIVATE virtual-input-core)
add_dependencies(azrael-virtual-input azrael-virtual-input-hook)
add_dependencies(azrael-virtual-input-selftest azrael-virtual-input-hook azrael-virtual-input-fixture)
foreach(target azrael-virtual-input-hook virtual-input-core azrael-virtual-input azrael-virtual-input-fixture azrael-virtual-input-selftest)
  target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8 /EHsc)
endforeach()
enable_testing()
add_test(NAME owned-classic-edit COMMAND azrael-virtual-input-selftest)
